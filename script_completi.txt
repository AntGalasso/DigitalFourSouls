## Human hotseat game scene (MP05 + MP06 + MP07). No art: Labels and Buttons only.
## Drives the SAME MatchController as the self-play test â€” game rules live
## there; this scene only renders state and translates clicks.
##
## MP07 hotseat flow: free action order â€” buy / attack / play loot in ANY
## order, each limited per turn (1 buy, 1 attack, 1 loot + 1 with the
## character effect) until "END TURN". Attacks are multi-roll sequences; every
## stack item (attack declared, roll, damage, monster death) opens a PRIORITY
## window in which ALL players respond in turn order starting from the active
## player â€” loot play and eternal items are valid responses (F). STOP attack
## no longer exists (D): an attack only ends when the monster or the player
## dies. Eternal modifiers (D6 / Belial) apply ONLY while the roll is pending,
## right after the roll and before it resolves (B). The shop has exactly 2
## slots refilled from the treasure deck (E). Boss kills become soul cards
## counting soul_value (Monstro = 2) towards the win condition at total 4 (J).
## Turn alternation P1->P2 is now enforced â€” the pass-device overlay starts
## the NEXT player, never the same one again (A).

const CARD_DIR := "res://resources/cards/base_set/"
const MatchControllerScript := preload("res://scripts/core/match_controller.gd")
const MONSTER_SLOTS := MatchControllerScript.MONSTER_SLOTS
const SHOP_SLOTS := MatchControllerScript.SHOP_SLOTS

var _controller
var _pending_action: String = ""    # "" | "attack" | "target:<hand index>" | "target_item"
var _next_player := 1               # who starts after "Pass the device" (MP07 A fix)
var _priority_open := false
var _priority_player := -1
var _priority_type := ""

var _turn_label: Label
var _hint_label: Label
var _player_labels: Array = []      # [Label P1, Label P2]
var _monster_buttons: Array = []    # [Button x MONSTER_SLOTS]

func _ready() -> void:
	_controller = MatchControllerScript.new()
	_controller.setup(2, Rng.rng.randi())
	_build_ui()
	EventBus.cents_changed.connect(func(_p, _d, _t): _refresh())
	EventBus.player_healed.connect(_on_player_healed)
	EventBus.player_damaged.connect(_on_player_damaged)
	EventBus.hp_restored.connect(_on_hp_restored)
	EventBus.monsters_restored.connect(_on_monsters_restored)
	EventBus.souls_changed.connect(func(_p, _s, _t): _refresh())
	EventBus.boss_defeated.connect(_on_boss_defeated)
	EventBus.monster_attacked.connect(_on_monster_attacked_log)
	EventBus.monster_damaged.connect(_on_monster_damaged_log)
	EventBus.monster_killed.connect(_on_monster_killed_log)
	EventBus.priority_pending.connect(_on_priority_pending)
	EventBus.priority_closed.connect(_on_priority_closed)
	EventBus.player_died.connect(_on_player_died)
	EventBus.eternal_used.connect(func(_p, _i, _r): _refresh())
	EventBus.card_drawn.connect(_on_card_drawn)
	EventBus.item_bought.connect(func(_p, _id): _refresh())
	EventBus.game_won.connect(_on_game_won)
	_controller.begin_turn(0)
	_refresh()
	_log_line("Welcome to Four Souls (prototype, %d cards). Free order: buy, ATTACK, play loot, use items â€” then END TURN." % _controller.total_cards_at_setup)

# --- Interaction --------------------------------------------------------------

func _on_monster_pressed(slot: int) -> void:
	var instance_id: String = _controller.table_monsters[slot]["instance_id"]
	if _pending_action == "attack":
		_pending_action = ""
		_refresh()
		var monster: Dictionary = _controller.table_monsters[slot]
		var monster_card: CardResource = monster["card"]
		var result: Dictionary = await _controller.attack(instance_id, GameState.current_player_index)
		match result.get("outcome", ""):
			"killed":
				_log_line("attack sequence: %s killed after %d rolls" % [monster_card.display_name, result["steps"]])
			"player_died":
				_log_line("attack sequence: you died after %d rolls" % result["steps"])
			_:
				if not result.get("valid", true):
					if result.has("busy"):
						_log_line("another action is still resolving â€” try again in a moment")
					elif result.get("limit", false):
						_log_line("attack limit reached (1 attack per turn)")
					else:
						_log_line("you can only attack on your own turn")
		_refresh()
	elif _pending_action.begins_with("target:"):
		var hand_index := int(_pending_action.substr(7))
		_pending_action = ""
		_play_hand_card(hand_index, instance_id)
	elif _pending_action == "target_item":
		_pending_action = ""
		_play_active_item(instance_id)
	_refresh()

func _on_shop_pressed(slot: int) -> void:
	var treasure: CardResource = _controller.get_shop_card(slot)
	if treasure != null and _controller.buy(slot, GameState.current_player_index):
		_log_line("bought %s for %dc" % [treasure.display_name, treasure.cost_cents])
	_refresh()

func _on_hand_pressed(hand_index: int) -> void:
	var card: CardResource = _card_from_hand(hand_index)
	if card == null:
		return
	var pid := GameState.current_player_index
	if _priority_open and _priority_player == pid:
		var target := ""
		if card.needs_target and not _controller.table_monsters.is_empty():
			target = _controller.table_monsters[0]["instance_id"]
		_respond({"action": "loot", "card_id": card.id, "target_id": target})
		return
	if card.needs_target and not _controller.table_monsters.is_empty():
		_pending_action = "target:%d" % hand_index
		_refresh()
		return
	_play_hand_card(hand_index, "")

func _play_hand_card(hand_index: int, target_id: String) -> void:
	var player_id := GameState.current_player_index
	var card: CardResource = _card_from_hand(hand_index)
	if card == null:
		return
	var payload := {}
	if target_id != "":
		payload = {"target_id": target_id}
	if not await _controller.play_loot(card, player_id, payload):
		_log_line("can't play %s â€” loot limit reached (1 per turn + character effect)" % card.display_name)
		return
	_log_line("played %s" % card.display_name)
	GameState.players[player_id]["hand"].erase(card.id)
	_refresh()

func _on_items_use(card: CardResource) -> void:
	var pid := GameState.current_player_index
	if _priority_open and _priority_player == pid:
		var target := ""
		if card.needs_target and not _controller.table_monsters.is_empty():
			target = _controller.table_monsters[0]["instance_id"]
		_respond({"action": "loot", "card_id": card.id, "target_id": target})
		return
	if card.needs_target and not _controller.table_monsters.is_empty():
		_pending_action = "target_item"
		_refresh()
		return
	_play_active_item("")

func _play_active_item(target_id: String) -> void:
	var player_id := GameState.current_player_index
	for card_id in GameState.players[player_id]["items_passive"]:
		var card: CardResource = _controller.card_resource(card_id)
		if card != null and card.playable:
			var payload := {}
			if target_id != "":
				payload = {"target_id": target_id}
			_log_line("used %s" % card.display_name)
			await _controller.play_loot(card, player_id, payload)
			_refresh()
			return

## RESOLVE / CONTINUE: closes the window when the ACTIVE player is asked.
func _on_resolve_pressed() -> void:
	_respond({"action": "resolve"})

# --- Priority window (MP07 F) -------------------------------------------------

## One stack item opened a priority window: if the ACTIVE player is asked the
## inline buttons appear (RESOLVE/CONTINUE + hand + eternal); if the OTHER
## player is asked, a pass-the-device overlay offers PASS + their loot and
## eternal item.
func _on_priority_pending(pid: int, stack_type: String, _payload: Dictionary) -> void:
	_priority_open = true
	_priority_player = pid
	_priority_type = stack_type
	if pid != GameState.current_player_index:
		_show_priority_overlay(pid)
	_refresh()

func _on_priority_closed(_stack_type: String, _payload: Dictionary) -> void:
	_priority_open = false
	_priority_player = -1
	_priority_type = ""
	_overlay.visible = false
	_refresh()

func _respond(response: Dictionary) -> void:
	if not _priority_open:
		return
	EventBus.priority_given.emit(_priority_player, response)

func _show_priority_overlay(pid: int) -> void:
	_overlay_label.text = "PRIORITY â€” P%d must respond" % (pid + 1)
	_overlay_button.visible = false
	for child in _overlay_box.get_children():
		child.queue_free()
	var pass_button := Button.new()
	pass_button.text = "PASS"
	pass_button.pressed.connect(_respond.bind({"action": "pass"}))
	_overlay_box.add_child(pass_button)
	_render_priority_eternal(pid)
	_render_priority_hand(pid)
	_overlay.visible = true

## The other player's eternal item, gated on the pending roll (MP07 B): D6 /
## Belial only while a roll is PENDING on the stack â€” never before, never
## after. Yum Heart (prevent_damage) answers any window: one button per
## possible player target.
func _render_priority_eternal(pid: int) -> void:
	var player := GameState.players[pid]
	var item: CardResource = _controller.card_resource(player["eternal_item_id"])
	if item == null:
		return
	var pending: bool = not _controller.get_pending_roll().is_empty()
	match item.eternal_action:
		"reroll_die":
			var d6 := Button.new()
			d6.text = "USE %s (reroll the pending roll)" % item.display_name
			d6.disabled = player["eternal_used"] or not pending
			d6.pressed.connect(_respond.bind({"action": "eternal", "payload": {}}))
			_overlay_box.add_child(d6)
		"adjust_die":
			var plus := Button.new()
			plus.text = "%s +1" % item.display_name
			plus.disabled = player["eternal_used"] or not pending
			plus.pressed.connect(_respond.bind({"action": "eternal", "payload": {"delta": 1}}))
			var minus := Button.new()
			minus.text = "%s -1" % item.display_name
			minus.disabled = player["eternal_used"] or not pending
			minus.pressed.connect(_respond.bind({"action": "eternal", "payload": {"delta": -1}}))
			_overlay_box.add_child(plus)
			_overlay_box.add_child(minus)
		"prevent_damage":
			for target in range(GameState.players.size()):
				var shield := Button.new()
				shield.text = "%s on P%d (prevent next damage)" % [item.display_name, target + 1]
				shield.disabled = player["eternal_used"]
				shield.pressed.connect(_respond.bind({"action": "eternal", "payload": {"target_id": target}}))
				_overlay_box.add_child(shield)

## The other player's hand, played as a priority response (their turn loot
## limit and character effect still apply in the engine).
func _render_priority_hand(pid: int) -> void:
	for card_id in GameState.players[pid]["hand"]:
		var card: CardResource = _controller.card_resource(card_id)
		if card == null:
			continue
		var target := ""
		if card.needs_target and not _controller.table_monsters.is_empty():
			target = _controller.table_monsters[0]["instance_id"]
		var button := Button.new()
		button.text = "PLAY %s" % card.display_name
		button.pressed.connect(_respond.bind({"action": "loot", "card_id": card_id, "target_id": target}))
		_overlay_box.add_child(button)

# --- Eternal items (MP06 g) ---------------------------------------------------

func _on_eternal_use(payload: Dictionary) -> void:
	var pid := GameState.current_player_index
	if _priority_open and _priority_player == pid:
		_respond({"action": "eternal", "payload": payload})
		return
	var result: Dictionary = _controller.use_eternal_item(pid, payload)
	if not result.get("ok", false):
		_log_line("eternal item unavailable: %s" % result.get("reason", "?"))
		return
	var item_name: String = _controller.card_resource(GameState.players[pid]["eternal_item_id"]).display_name
	if result.has("new_roll") and result.has("old_roll"):
		_log_line("%s rerolls %d -> %d" % [item_name, result["old_roll"], result["new_roll"]])
	elif result.has("new_roll"):
		_log_line("%s adjusts the roll to %d" % [item_name, result["new_roll"]])
	elif result.has("target_player"):
		_log_line("%s on P%d: next damage this turn is prevented" % [item_name, result["target_player"] + 1])
	elif result.has("target_instance"):
		_log_line("%s on %s: next damage is prevented" % [item_name, result["target_instance"]])
	_refresh()

## Turn alternation fix (MP07 A): remember who starts next, and let the
## pass-device overlay begin THAT player â€” never the same one again.
func _on_end_turn() -> void:
	_next_player = _controller.end_turn()
	_show_overlay("Pass the device to P%d" % (_next_player + 1), "Start P%d's turn" % (_next_player + 1))

func _on_overlay_button() -> void:
	var pid := _next_player
	_controller.begin_turn(pid)
	_next_player = (pid + 1) % GameState.players.size()
	_overlay.visible = false
	_refresh()

func _show_overlay(text: String, button_text: String) -> void:
	_overlay_label.text = text
	_overlay_button.text = button_text
	_overlay_button.visible = true
	for child in _overlay_box.get_children():
		child.queue_free()
	_overlay.visible = true

## Choice overlay (e.g. "choose an item to lose"): replaces the single button
## with one button per option; the single button is hidden.
func _show_choice_overlay(text: String, options: Array, on_choose: Callable) -> void:
	_overlay_label.text = text
	_overlay_button.visible = false
	for child in _overlay_box.get_children():
		child.queue_free()
	for option in options:
		var button := Button.new()
		button.text = str(option)
		button.pressed.connect(on_choose.bind(option))
		_overlay_box.add_child(button)
	_overlay.visible = true

# --- State rendering ----------------------------------------------------------

func _refresh() -> void:
	var pid := GameState.current_player_index
	_turn_label.text = "TURN â€” P%d" % (pid + 1)
	for i in range(2):
		var p := GameState.players[i]
		var active := " (your turn)" if i == pid else ""
		var char_name: String = p["character_id"].trim_prefix("character_").capitalize()
		var char_state := "effect ready" if not p["character_effect_used"] else "effect used"
		_player_labels[i].text = "P%d%s â€” %s\n  HP %d/%d   cents %d   souls %s (value %d)\n  items: %s\n  %s (eternal: %s)" % [
			i + 1, active, char_name, p["hp"], p["max_hp"], p["cents"], str(p["souls"]),
			GameState.soul_total(i), str(p["items_passive"]), char_state, p["eternal_item_id"].trim_prefix("eternal_")]
	for slot in range(MONSTER_SLOTS):
		var button: Button = _monster_buttons[slot]
		if slot < _controller.table_monsters.size():
			var monster: Dictionary = _controller.table_monsters[slot]
			var monster_card: CardResource = monster["card"]
			var boss_tag := " â€” BOSS" if monster_card.is_boss else ""
			button.text = "%s%s\n die %d â€” hp %d" % [monster_card.display_name, boss_tag, monster_card.die_value, monster["hp"]]
			button.tooltip_text = monster_card.effect_description
			button.disabled = _priority_open
		else:
			button.text = "empty slot"
			button.disabled = true
	var player := GameState.players[pid]
	var buy_limit: bool = player["buys_used"] >= 1
	for i in range(SHOP_SLOTS):
		var button: Button = _shop_buttons[i]
		var treasure: CardResource = _controller.get_shop_card(i)
		if treasure == null:
			button.text = "empty shop"
			button.disabled = true
		elif _priority_open:
			button.text = "%s" % treasure.display_name
			button.disabled = true
		elif buy_limit:
			button.text = "%s â€” buy limit used" % treasure.display_name
			button.disabled = true
		elif _controller.is_bought(treasure.id, pid):
			button.text = "%s â€” bought" % treasure.display_name
			button.disabled = true
		elif player["cents"] < treasure.cost_cents:
			button.text = "%s (%dc) â€” too poor" % [treasure.display_name, treasure.cost_cents]
			button.disabled = true
		else:
			button.text = "%s (%dc)" % [treasure.display_name, treasure.cost_cents]
			button.disabled = false
		button.tooltip_text = treasure.description
	_rebuild_hand()
	_rebuild_items()
	_rebuild_eternal()
	_update_hint()
	_attack_button.disabled = _priority_open or player["attacks_used"] >= 1
	_resolve_button.visible = _priority_open and _priority_player == pid
	if _priority_open and _priority_player == pid:
		_resolve_button.text = "RESOLVE roll" if _priority_type == "roll" else "CONTINUE"
	_cancel_button.disabled = _pending_action == "" or _priority_open
	_end_button.disabled = _priority_open

func _rebuild_hand() -> void:
	for child in _hand_box.get_children():
		child.queue_free()
	var player := GameState.players[GameState.current_player_index]
	var loot_limit: bool = player["loot_played"] >= 1 and player["character_effect_used"]
	var hand: Array = player["hand"]
	for i in range(hand.size()):
		var card: CardResource = _controller.card_resource(hand[i])
		if card == null:
			continue
		var button := Button.new()
		button.text = "%s â€” %s" % [card.display_name, _card_summary(card)]
		button.tooltip_text = card.text
		button.pressed.connect(_on_hand_pressed.bind(i))
		if loot_limit:
			button.text += " (limit)"
			button.disabled = true
		_hand_box.add_child(button)

func _rebuild_items() -> void:
	for child in _items_box.get_children():
		child.queue_free()
	var items: Array = GameState.players[GameState.current_player_index]["items_passive"]
	for card_id in items:
		var card: CardResource = _controller.card_resource(card_id)
		if card != null and card.playable:
			var button := Button.new()
			button.text = "USE %s â€” %s" % [card.display_name, _card_summary(card)]
			button.tooltip_text = card.description
			button.pressed.connect(_on_items_use.bind(card))

## STEP H: eternal items work on the PENDING attack roll (MP07 B) â€” D6 rerolls
## and Belial Book adjusts by -1; the FINAL roll (what the attack resolves and
## monster_attacked carries) is the modified one. Recharge at end of the
## owner's turn and at the owner's next turn start. Two attacks, one modifier
## each: +5 bonus vs die 5 on an hp-1 monster guarantees the first roll kills,
## so final_roll IS the modified roll â€” no dice luck involved.
func _check_eternal_items() -> void:
	print("\n-- CHECK: eternal items on the pending roll (STEP H) --")
	var p0: Dictionary = GameState.players[0]
	var p1: Dictionary = GameState.players[1]
	p0["character_id"] = "character_isaac"
	p0["eternal_item_id"] = "eternal_d6"

	var steven: CardResource = load(CARD_DIR + "treasure_steven.tres")
	if fly == null or steven == null:
		Log.error("turn_loop_test: monster_fly / treasure_steven missing (run the import)")
		get_tree().quit(1)
		return
	for i in range(5):
		CardEffectRegistry.register_card(steven, 0, "treasure_steven#h%d" % i)
	_controller.begin_turn(0)
	## Attack 1: Belial Book -1 only â€” roll_after is roll_before - 1, always.
	_check_roll_before = -1
	_check_roll_after = -1
	_check_belial_ok = false
	_check_first_attacked_roll = -1
	_check_roll_checked = false
	var handler_belial := func(pid: int, stack_type: String, _payload: Dictionary):
		if stack_type == "pending_roll" and not _check_roll_checked:

			_check_belial_ok = _controller.use_eternal_item(1, {"delta": -1}).get("ok", false)
			_check_roll_after = _controller.get_pending_roll().get("roll", -1)
		EventBus.priority_given.emit.call_deferred(pid, {"action": "resolve" if stack_type == "pending_roll" else "pass"})
	var on_attacked := func(_pid: int, _iid: String, roll: int, _bonus: int, _die: int):
		if _check_first_attacked_roll == -1:
			_check_first_attacked_roll = roll  # FIRST emission = the modified roll
	_controller.table_monsters.clear()

	EventBus.monster_attacked.disconnect(on_attacked)
	var ok: bool = _check_belial_ok
	ok = ok and _check_roll_after == _check_roll_before - 1   # Belial -1 applied while pending
	ok = ok and _check_first_attacked_roll == _check_roll_after
	ok = ok and result_a.get("final_roll", -1) == _check_roll_after
	## Attack 2: D6 reroll only â€” the reroll is DEFERRED to modifier resolution,

			_check_roll_checked = true
			var res: Dictionary = _controller.use_eternal_item(0, {})
			_check_d6_ok = res.get("ok", false) and res.get("reason", "") != "no_pending_roll"
		EventBus.priority_given.emit.call_deferred(pid, {"action": "resolve" if stack_type == "pending_roll" else "pass"})
	var on_attacked2 := func(_pid: int, _iid: String, roll: int, _bonus: int, _die: int):
		if _check_second_attacked_roll == -1:
			_check_second_attacked_roll = roll  # FIRST emission = the rerolled value
	_controller.table_monsters.clear()
	_controller.table_monsters.append({"instance_id": "monster_fly#h2", "card": fly, "hp": 1})
	_controller.begin_turn(0)  # the attack limit is 1 per turn; fresh turn for the 2nd attack

	EventBus.priority_pending.connect(handler_d6)

	var result_b: Dictionary = await _controller.attack("monster_fly#h2", 0)

	_check_d6_new = _check_second_attacked_roll
	ok = ok and _check_d6_ok and _check_d6_new > 0

	ok = ok and not p1["eternal_used"]
	_controller.end_turn()

		"PASS" if ok else "FAIL", _check_roll_before, _check_roll_after, _check_d6_new])
	_record_check("STEP H eternal items on the pending roll (B)", ok)

## Duck-typed stand-in for Rng.rng: replays scripted die results in order
## (default 1 once dry) so the example sequences are byte-exact, not lucky.
class FakeRng extends RefCounted:
	var queue: Array = []
	func randi_range(_from: int, _to: int) -> int:
		if queue.is_empty():
			return 1
		return queue.pop_front()

func _install_fake_rng(queue: Array) -> void:
	if _real_rng == null:
		_real_rng = Rng.rng
	var fake := FakeRng.new()
	fake.queue = queue.duplicate()
	Rng.rng = fake

func _restore_rng() -> void:
	if _real_rng != null:
		Rng.rng = _real_rng
	_real_rng = null

## Replaces the whole table with ONE fresh monster instance.
func _set_single_monster(instance_id: String, card: CardResource, hp: int) -> void:
	for m in _controller.table_monsters:
		CardEffectRegistry.unregister_monster(m["instance_id"])
	_controller.table_monsters.clear()
	_controller.table_monsters.append({"instance_id": instance_id, "card": card, "hp": hp})

func _find_table_hp(instance_id: String) -> int:
	for m in _controller.table_monsters:
		if m["instance_id"] == instance_id:
			return m["hp"]
	return -999

extends Node
## GameState (autoload/singleton)
##
## Single source of truth for the current match. Phase 1 (local hotseat)
## keeps this dead simple. Phase 2 (networked) will make this authoritative
## on the host and mirror it to peers via RPCs â€” deliberately NOT designed
## yet, don't add networking code here until the local loop is proven.

var players: Array[Dictionary] = []     # see PLAYER_TEMPLATE below
var current_player_index: int = 0
var current_phase: String = "buy"
var turn_number: int = 0
var soul_values: Dictionary = {}        # soul name -> value (1 by default, bosses 1-2 per MP07 J)
## Active shields (ESEMPIO 4): target_player_id -> {owner, item_name}.
## Yum Heart is DAMAGE PREVENTION, not a heal; the flag on the player dict is
## the fast check, this map tracks who applied it and when it expires.
var shields: Dictionary = {}

const PLAYER_TEMPLATE := {
	"id": -1,
	"character_id": "",        # set by MatchController at setup (random draw)
	"eternal_item_id": "",     # the character's eternal item (never lost to the death penalty)
	"hp": 2,
	"max_hp": 2,               # can permanently grow by healing at full HP
	"damage": 1,
	"cents": 3,
	"souls": [],       # Array[String] soul NAMES, e.g. ["monster_monstro", "blue"] (MP07 J)
	"hand": [],         # Array[String] card ids
	"items_active": [], # Array[String] card ids
	"items_passive": [],# Array[String] card ids
	"attacks_used": 0,  # per own turn (reset in begin_turn)
	"buys_used": 0,
	"loot_played": 0,
	"character_effect_used": false,  # recharges at the start of the owner's turn
	"eternal_used": false,           # recharges at the owner's turn start (+ turn end for D6/Belial)
	"shield_next_damage": false,   # Yum Heart: prevents next damage instance this turn
}

func reset_match(player_count: int) -> void:
	players.clear()
	soul_values.clear()
	shields.clear()
	for i in range(player_count):
		var p := PLAYER_TEMPLATE.duplicate(true)
		p["id"] = i
		players.append(p)
	current_player_index = 0
	current_phase = "buy"
	turn_number = 0

func get_current_player() -> Dictionary:
	return players[current_player_index]

## Shared active_player_id (ESEMPIO 1): called ALWAYS at the end of the
## ending phase â€” the step that must never be missing. Logs and emits so
## drivers/UI can re-read GameState.current_player_index afterwards.
func advance_active_player() -> int:
	if players.is_empty():
		return current_player_index
	var previous := current_player_index
	current_player_index = (current_player_index + 1) % players.size()
	turn_number += 1
	print("end of turn: active player = P%d" % (current_player_index + 1))
	EventBus.active_player_advanced.emit(previous, current_player_index)
	return current_player_index

# --- Mutations -------------------------------------------------------------
# Effects and the turn loop mutate match state ONLY through these helpers,
# so GameState (state) and EventBus (notifications) never drift apart.

func gain_cents(player_id: int, amount: int) -> void:
	var player := _get_player(player_id)
	if player.is_empty():
		return
	player["cents"] += amount
	EventBus.cents_changed.emit(player_id, amount, player["cents"])

func lose_cents(player_id: int, amount: int) -> void:
	gain_cents(player_id, -amount)

## Gives the player a soul card. A boss soul is ONE entry in the souls list
## (its NAME) that counts `value` towards the 4-soul win condition (MP07 J).
## Every souls change re-checks the win condition immediately.
func gain_soul(player_id: int, soul_type: String, value: int = 1) -> void:
	var player := _get_player(player_id)
	if player.is_empty():
		return
	player["souls"].append(soul_type)
	soul_values[soul_type] = value
	var total := soul_total(player_id)
	EventBus.soul_gained.emit(player_id, soul_type, total)
	EventBus.souls_changed.emit(player_id, player["souls"], total)
	var winner := has_winner()
	if winner >= 0:
		EventBus.game_won.emit(winner)

## Sum of the player's soul VALUES (bosses count 2, normal souls 1).
func soul_total(player_id: int) -> int:
	var player := _get_player(player_id)
	if player.is_empty():
		return 0
	var total := 0
	for soul_name in player["souls"]:
		total += soul_values.get(soul_name, 1)
	return total

func damage_player(player_id: int, amount: int, source_id: String) -> void:
	var player := _get_player(player_id)
	if player.is_empty():
		return
	if consume_shield(player_id, amount):
		EventBus.player_damaged.emit(player_id, 0, source_id)  # fully prevented
		return
	player["hp"] -= amount
	EventBus.player_damaged.emit(player_id, amount, source_id)

# --- Shields (ESEMPIO 4): prevention flags, never heals ----------------------

## Yum Heart activation, step 2: "apply_shield(target, flag=shield_next_damage,
## source=yum_heart, expires=end_of_turn:<owner>)". One call does everything â€”
## there is no follow-up menu and no quantity.
func apply_shield(target_id: int, owner_id: int, item_name: String) -> void:
	var player := _get_player(target_id)
	if player.is_empty():
		return
	player["shield_next_damage"] = true
	shields[target_id] = {"owner": owner_id, "item_name": item_name}
	EventBus.shield_applied.emit(target_id, item_name, owner_id)

## Called by damage_player when the flag is up: the WHOLE instance is
## cancelled, the flag is removed after consumption.
func consume_shield(target_id: int, absorbed_amount: int) -> bool:
	if not shields.has(target_id):
		return false
	var info: Dictionary = shields[target_id]
	shields.erase(target_id)
	var player := _get_player(target_id)
	if not player.is_empty():
		player["shield_next_damage"] = false
	print("P%d's shield (%s) absorbs the damage, 0 hp lost" % [target_id + 1, info.get("item_name", "?")])
	EventBus.shield_consumed.emit(target_id, absorbed_amount)
	return true

## Ending phase of the OWNER's turn (expires="end_of_turn:<owner>"): every
## shield they handed out that was never consumed is removed. Returns the
## target ids whose shields expired so the controller can recharge the item.
func expire_shields_owned_by(owner_id: int) -> Array:
	var expired: Array = []
	for target_id in shields.keys():
		if shields[target_id].get("owner", -1) == owner_id:
			var info: Dictionary = shields[target_id]
			expired.append(target_id)
			shields.erase(target_id)
			var player := _get_player(target_id)
			if not player.is_empty():
				player["shield_next_damage"] = false
			print("%s shield on P%d expires unused" % [info.get("item_name", "?"), target_id + 1])
			EventBus.shield_expired.emit(target_id)
	return expired

## Heals a player. Healing beyond max HP permanently increases max HP (MP06 g):
## a heal at full HP still nets +1 HP, exactly like the official Yum Heart rule.
func heal(player_id: int, amount: int) -> void:
	var player := _get_player(player_id)
	if player.is_empty():
		return
	if amount <= 0:
		return
	var new_hp: int = player["hp"] + amount
	if new_hp > player["max_hp"]:
		player["max_hp"] = new_hp
	player["hp"] = new_hp
	EventBus.player_healed.emit(player_id, amount, player["hp"])

## End-of-turn full restore (MP06 d): sets HP to max WITHOUT the permanent
## increase rule â€” only card effects grow max HP, not the turn-end heal.
func restore_hp(player_id: int) -> void:
	var player := _get_player(player_id)
	if player.is_empty():
		return
	player["hp"] = player["max_hp"]
	EventBus.hp_restored.emit(player_id, player["hp"])

func _get_player(player_id: int) -> Dictionary:
	for p in players:
		if p["id"] == player_id:
			return p
	return {}

## Win condition (MP07 J): the SUM of soul values reaches 4 (a soul_value 2
## boss counts double, still ONE soul card in the list).
func has_winner() -> int:
	for p in players:
		if soul_total(p["id"]) >= 4:
			return p["id"]
	return -1

	return int(rng.randi_range(1, 6))

func shuffle(items: Array) -> void:
	for i in range(items.size() - 1, 0, -1):
		var j: int = int(rng.randi_range(0, i))
		var tmp = items[i]
		items[i] = items[j]
		items[j] = tmp

extends Node
## CardEffectRegistry (autoload/singleton)
##
## Two card behavior paths, decided per card (never both):
##
##  DATA path (default): CardResource.simple_effect is a keyword applied by
##  _apply_simple_effect(), e.g. "gain_cents" / "gain_hp" / "damage_monster".
##  Trivial cards (the majority of a 300-card set) need nothing but a CSV row.
##  Data-driven passives use trigger_events the same way scripted ones do.
##
##  SCRIPT path (escape hatch): CardResource.effect_script_path points at a
##  CardEffect subclass for cards that need conditions, sequencing or choices.
##  The registry instantiates it, wires trigger_events to EventBus signals,
##  and dispatches on_play / on_attacked / on_killed when the turn loop plays
##  a card or fights a monster.
##
## Every card that enters play (bought treasure, drawn monster) gets an
## instance_id from the turn loop, so multiple copies of the same card never
## collide â€” _by_card and _owned are keyed by instance_id, not card_id.
## Effects are unmounted (signals disconnected, instance dropped) when their
## card leaves the game, so _effects does not grow without bound.

var _effects: Array = []                       # every live CardEffect instance
var _by_card: Dictionary = {}                  # instance_id -> CardEffect (scripted owned cards)
var _owned: Dictionary = {}                    # player_id -> {instance_id -> CardResource}
var _owned_effects: Dictionary = {}            # player_id -> Array[CardEffect] (scripted owned)
var _monster_by_instance: Dictionary = {}      # monster instance_id -> CardEffect
var _trigger_connections: Array = []           # [{instance_id, signal_name, callable, effect, player_id}]
## Connections are NOT attached to EventBus directly: one dispatcher per signal
## re-emits to them in _policy_order() (active player first, then clockwise,
## same player in registration order) â€” the deterministic default for
## simultaneous effects (Metaprompt 05, decision c).

func _ready() -> void:
	EventBus.turn_started.connect(_dispatch_turn_started)
	EventBus.card_played.connect(_dispatch_card_played)
	EventBus.monster_killed.connect(_dispatch_monster_killed)

## Registers a card owned by a player (bought treasure): stores it, and if
## scripted, instantiates its effect + subscribes trigger_events.
## instance_id must be unique per copy (the loop assigns it).
func register_card(card: CardResource, player_id: int, instance_id: String) -> void:
	if card == null:
		return
	var owned: Dictionary = _owned.get(player_id, {})
	owned[instance_id] = card
	_owned[player_id] = owned
	if card.effect_script_path != "":
		var effect := _instantiate(card, player_id)
		if effect == null:
			return
		_by_card[instance_id] = effect
		var owned_effects: Array = _owned_effects.get(player_id, [])
		owned_effects.append(effect)
		_owned_effects[player_id] = owned_effects
		for signal_name in card.trigger_events:
			_wire_trigger(effect, instance_id, signal_name)
	elif card.simple_effect != "":
		for signal_name in card.trigger_events:
			_wire_data_trigger(card, player_id, instance_id, signal_name)

## Removes an owned card copy (discarded / leaves game): unmounts its effect
## and disconnects every trigger wired for that instance.
func unregister_card(player_id: int, instance_id: String) -> void:
	var owned: Dictionary = _owned.get(player_id, {})
	owned.erase(instance_id)
	var effect: CardEffect = _by_card.get(instance_id)

## Deterministic default order for simultaneous effects: the ACTIVE player's
## connections first, then the other players clockwise, and within one player
## in registration (= purchase) order. No human choice of order for now.

func _policy_order() -> Array:
	var count := GameState.players.size()
	if count == 0:
		return _trigger_connections.duplicate()
	var active := GameState.current_player_index
	var ordered: Array = []
	for i in range(count):
		var pid := (active + i) % count
		for conn in _trigger_connections:
			if conn["player_id"] == pid:
				ordered.append(conn)
	return ordered

	for conn in _policy_order():
		if conn["signal_name"] == "turn_started":
			conn["callable"].call(player_id)

	for conn in _policy_order():

func _dispatch_monster_killed(player_id: int, instance_id: String, rewards: Dictionary) -> void:
	for conn in _policy_order():
		if conn["signal_name"] == "monster_killed":
			conn["callable"].call(player_id, instance_id, rewards)

class_name EventStack
extends RefCounted
## The real event/stack queue (MP07 build).
##
## Typed LIFO entries, exactly the three kinds from the metaprompt examples:
##   {"type": "pending_roll",  "id": "r1", "owner": pid, "target": instance_id,
##    "value": int, "bonus": int, "die_value": int}
##   {"type": "pending_effect","id": "e1", "owner": pid, "card_id": String,
##    "payload": Dictionary}                       (loot played from hand)
##   {"type": "modifier",      "id": "m1", "owner": pid, "target_id": "r1",
##    "effect": "reroll"|"adjust", "item_id"/"item_name", ["delta": +-1,
##    "applied": true]}                            (D6 / Belial Book responses)
##
## resolve_top() applies ONLY the top entry, and only after the priority
## window closed (every player passed consecutively starting from the active
## player â€” the window itself lives in MatchController._run_stack_window).
## Resolution is delegated to the `resolver` Callable owned by the
## controller; this class only owns ORDER and CASCADE: after a modifier
## resolves, resolution continues down onto its target pending_roll without
## reopening a window (ESEMPIO 2 step 3: resolve the modifier, then resolve
## r1 with its updated value).

signal changed()

var entries: Array = []            # LIFO â€” last element is the top
var resolver: Callable = Callable()  # func(entry: Dictionary) -> Dictionary
var _counters := {"r": 0, "e": 0, "m": 0}

func push(entry: Dictionary) -> Dictionary:
	var kind: String = entry.get("type", "?")
	var prefix := "r"
	match kind:
		"pending_roll": prefix = "r"
		"pending_effect": prefix = "e"
		"modifier": prefix = "m"
		_: prefix = "x"
	_counters[prefix] = _counters.get(prefix, 0) + 1
	entry["id"] = "%s%d" % [prefix, _counters[prefix]]
	entries.append(entry)
	EventBus.stack_entry_pushed.emit(entry.duplicate())
	changed.emit()
	return entry

func pop() -> Dictionary:
	if entries.is_empty():
		return {}
	var entry: Dictionary = entries.pop_back()
	changed.emit()
	return entry

func top() -> Dictionary:
	if entries.is_empty():
		return {}
	return entries[-1]

## By-reference lookup: mutating the returned Dictionary mutates the entry.
func find(id: String) -> Dictionary:
	for entry in entries:
		if entry.get("id", "") == id:
			return entry
	return {}

func top_of_type(type: String) -> Dictionary:
	for i in range(entries.size() - 1, -1, -1):
		if entries[i].get("type", "") == type:
			return entries[i]
	return {}

func size() -> int:
	return entries.size()

func is_empty() -> bool:
	return entries.is_empty()

func clear(reason: String = "") -> void:
	entries.clear()
	EventBus.stack_cleared.emit(reason)
	changed.emit()

## Resolves ONE batch from the top: pops the top entry, applies it through
## the resolver, then cascades down while modifiers sit above their target
## pending_roll (LIFO). Returns the result of the LAST entry applied â€” for a
## modifier chain that is the pending_roll's hit/miss result (ESEMPIO 2/2b).
func resolve_top() -> Dictionary:
	var first := pop()
	var result: Dictionary = {}
	if first.is_empty():
		return result
	result = await resolver.call(first)
	EventBus.stack_entry_resolved.emit(first.duplicate(), result.duplicate())
	# Cascade: any further modifiers on top were answered before the window
	# closed; they resolve now, then hand resolution to their target roll.
	while not is_empty() and top().get("type", "") == "modifier":
		var modifier := pop()
		await resolver.call(modifier)
		EventBus.stack_entry_resolved.emit(modifier.duplicate(), {})
	if first.get("type", "") == "modifier" and not is_empty() and top().get("type", "") == "pending_roll":
		var roll_entry := pop()
		result = await resolver.call(roll_entry)
		EventBus.stack_entry_resolved.emit(roll_entry.duplicate(), result.duplicate())
	return result

		return false
	if player["buys_used"] >= 1:
		return false
	if player["cents"] < card.cost_cents:
		return false
	var bought: Dictionary = _bought.get(player_id, {})
	if bought.has(card.id):
		return false
	GameState.lose_cents(player_id, card.cost_cents)
	player["items_passive"].append(card.id)
	bought[card.id] = true
	_bought[player_id] = bought
	player["buys_used"] += 1
	_instance_counter += 1
	var instance_id := "%s#%d" % [card.id, _instance_counter]
	shop_slots.remove_at(slot_index)
	_refill_shop()
	EventBus.item_bought.emit(player_id, card.id)
	CardEffectRegistry.register_card(card, player_id, instance_id)
	return true

## Attacks the monster with the given instance_id (ESEMPI 2, 2b, 3). Each
## roll is PUSHED on the stack as a pending_roll, the priority window opens
## to ALL players (generic modifiers like D6/Belial may answer), then the
## stack resolves top-first: modifier(s) cascade onto the pending_roll which
## resolves into HIT/MISS. HIT -> apply_damage -> check_death -> rewards from
## card data -> discard monster card -> refill slot; MISS -> the attacker
## takes 1 damage (no voluntary stop, MP07 D). reset_combat_state() runs on
## EVERY exit path. Returns {"valid", "outcome": "killed"|"player_died",
## "steps", "rolls", "final_roll"}.
func attack(instance_id: String, player_id: int) -> Dictionary:
	var monster := _find_monster(instance_id)
	if monster.is_empty():
		return {"valid": false}
	if _combat_in_progress or _effect_window_open:
		return {"valid": false, "busy": true}
	if player_id != GameState.current_player_index:
		return {"valid": false}
	var player := GameState.players[player_id]
	if player["attacks_used"] >= 1:
		return {"valid": false, "limit": true}
	player["attacks_used"] += 1
	_combat_in_progress = true
	var steps := 0
	var outcome := ""
	var rolls: Array = []
	var final_roll := 0
	while true:
		var m := _find_monster(instance_id)
		if m.is_empty():
			outcome = "killed"  # removed by an effect during the window
			break
		steps += 1
		var monster_card: CardResource = m["card"]
		# ESEMPIO 2 step 1: raw roll -> PUSH {type:"pending_roll", ...}
		var raw_roll := Rng.roll_die()
		var bonus := CardEffectRegistry.get_roll_bonus(player_id)
		var roll_entry: Dictionary = stack.push({
			"type": "pending_roll", "owner": player_id, "target": instance_id,
			"value": raw_roll + bonus, "bonus": bonus, "die_value": monster_card.die_value,
			"source": "attack"})
		print("%s rolls %d (pending, not yet resolved)" % [_pname(player_id), raw_roll + bonus])
		# ESEMPIO 2 step 2: priority window â€” ALL players in turn order,
		# consecutive passes close it; answers push generic modifiers.
		await _run_stack_window(roll_entry)
		# ESEMPIO 2 step 3: resolution from the top (modifier -> pending_roll).
		var result: Dictionary = await stack.resolve_top()
		var roll: int = result.get("roll", roll_entry.get("value", 0))
		final_roll = roll
		rolls.append(roll)
		if result.get("hit", false):
			# ESEMPIO 3 step 1: damage from the ATTACKER's stats, never hardcoded.
			var damage: int = GameState.players[player_id]["damage"] + CardEffectRegistry.get_damage(player_id)
			EventBus.monster_damaged.emit(instance_id, damage)
			var damaged := _find_monster(instance_id)
			# ESEMPIO 3 step 2: check_death -> resolve_monster_death (rewards,
			# souls/win check, discard, refill all hang off monster_killed).
			if damaged.is_empty() or damaged["hp"] <= 0:
				outcome = "killed"
				EventBus.monster_killed.emit(player_id, instance_id, _kill_rewards(m["card"]))
				break
			# survived: the sequence continues (MP07 D â€” no voluntary stop)
		else:
			if await _run_player_damage(player_id, 1, instance_id):
				outcome = "player_died"
				break
	reset_combat_state()
	return {"valid": true, "outcome": outcome, "steps": steps, "rolls": rolls, "final_roll": final_roll}

## A miss hurts the attacker; if HP reaches 0 they die mid-attack and the
## death flow (penalty -> ending phase -> advance) completes before the
## attack returns. Returns true when the player died.
func _run_player_damage(player_id: int, amount: int, source_id: String) -> bool:
	GameState.damage_player(player_id, amount, source_id)
	if GameState.players[player_id]["hp"] <= 0:
		print("  >>> %s dies mid-attack" % _pname(player_id))
		await die(player_id)
		return true
	return false

## Plays a card from hand (loot or active item) with an optional target
## (ESEMPIO 5). The play becomes a PENDING_EFFECT entry on the stack, the
## priority window opens to ALL players (even non-active ones), then the
## effect resolves. Loot limits are charged synchronously BEFORE the window.
## Returns false synchronously when the play is refused; true once resolved.
func play_loot(card: CardResource, player_id: int, payload: Dictionary = {}) -> bool:
	if card == null:
		return false
	if _effect_window_open or _combat_in_progress:
		return false
	if card.type == CardResource.CardType.LOOT:
		var player := GameState.players[player_id]
		if player_id == GameState.current_player_index:
			if player["loot_played"] < 1:
				player["loot_played"] += 1
			elif _character_effect_is_loot(player_id) and not player["character_effect_used"]:
				player["character_effect_used"] = true
				print("  %s uses character effect to play an extra loot" % _pname(player_id))
			else:
				return false
		else:
			if not _character_effect_is_loot(player_id) or player["character_effect_used"]:
				return false
			player["character_effect_used"] = true
			print("  %s uses character effect to play loot on another turn" % _pname(player_id))
	_effect_window_open = true
	# ESEMPIO 5 step 1: PUSH {type:"pending_effect", id:"e1", card, target}
	var entry: Dictionary = stack.push({
		"type": "pending_effect", "owner": player_id,
		"card_id": card.id, "payload": payload.duplicate()})
	var target_line := ""
	var target_id: String = payload.get("target_id", "")
	if target_id != "":
		var target := _find_monster(target_id)
		target_line = " targeting %s" % (target["card"].display_name if not target.is_empty() else target_id)
	print("%s plays %s%s (pending)" % [_pname(player_id), card.display_name, target_line])
	# ESEMPIO 5 step 2: priority window visible to EVERY player...
	await _run_stack_window(entry)
	# ...step 3: nobody responded (consecutive passes) -> e1 resolves.
	await stack.resolve_top()
	_effect_window_open = false
	return true

## Uses the player's eternal item as GENERIC stack answers (ESEMPI 2, 2b, 4):
##  - D6 (reroll_die): requires a pending_roll ON TOP of the stack; pushes a
##    "modifier" entry â€” the actual reroll happens at MODIFIER RESOLUTION
##    ("D6 reroll: r1 value changes from X to Y").
##  - Belial Book (adjust_die): requires a pending_roll on top; adjusts the
##    roll value IMMEDIATELY at activation ("uses Belial Book (+1): rX value
##    3 -> 4"); the pushed modifier records it for the cascade.
##  - Yum Heart (prevent_damage): PREVENTION SHIELD on a player (or monster),
##    never a heal; menu closes in the same call.
## There is no "modify before rolling" â€” no pending_roll, no action (B).
func use_eternal_item(player_id: int, payload: Dictionary = {}) -> Dictionary:
	var player := GameState.players[player_id]
	if player.is_empty():
		return {"ok": false}
	if player["eternal_used"]:
		return {"ok": false, "reason": "recharging"}
	var item_id: String = player["eternal_item_id"]
	if item_id == "":
		return {"ok": false, "reason": "no_eternal_item"}
	var item: CardResource = card_resource(item_id)
	if item == null:
		return {"ok": false, "reason": "missing_card"}
	var result := {"ok": true, "item_id": item_id}
	match item.eternal_action:
		"reroll_die":
			var roll_entry: Dictionary = stack.top_of_type("pending_roll")
			if roll_entry.is_empty():
				return {"ok": false, "reason": "no_pending_roll"}
			stack.push({
				"type": "modifier", "effect": "reroll", "item_id": item.id,
				"item_name": item.display_name, "owner": player_id,
				"target_id": roll_entry["id"]})
			# ESEMPIO 2 step 2b: "P2 activates D6 targeting r1"
			print("%s activates %s targeting %s" % [_pname(player_id), item.display_name, roll_entry["id"]])
			result["target_roll"] = roll_entry["id"]
		"adjust_die":
			var roll_entry: Dictionary = stack.top_of_type("pending_roll")
			if roll_entry.is_empty():
				return {"ok": false, "reason": "no_pending_roll"}
			var delta: int = clamp(int(payload.get("delta", 1)), -1, 1)
			var old_value: int = roll_entry["value"]
			roll_entry["value"] = old_value + delta  # immediate (ESEMPIO 2b step 2)
			stack.push({
				"type": "modifier", "effect": "adjust", "delta": delta,
				"applied": true, "item_id": item.id, "item_name": item.display_name,
				"owner": player_id, "target_id": roll_entry["id"]})
			print("%s uses %s (%+d): %s value %d -> %d" % [
				_pname(player_id), item.display_name, delta, roll_entry["id"], old_value, roll_entry["value"]])
			result["new_roll"] = roll_entry["value"]
		"prevent_damage":
			var target = payload.get("target_id", -1)
			if typeof(target) == TYPE_STRING:
				if _find_monster(target).is_empty():
					return {"ok": false, "reason": "no_target"}
				_monster_shields[target] = {"owner": player_id, "item_name": item.display_name}
				print("%s uses %s on %s: next damage to %s this turn is prevented" % [
					_pname(player_id), item.display_name, target, target])
				result["target_instance"] = target
			else:
				var tid: int = target
				if tid < 0 or tid >= GameState.players.size():
					return {"ok": false, "reason": "no_target"}
				# ESEMPIO 4 step 2 â€” one call, menu closes here, no quantities.
				GameState.apply_shield(tid, player_id, item.display_name)
				print("%s uses %s on %s: next damage to %s this turn is prevented" % [
					_pname(player_id), item.display_name, _pname(tid), _pname(tid)])
				result["target_player"] = tid
		_:
			return {"ok": false, "reason": "unknown_action"}
	player["eternal_used"] = true
	EventBus.eternal_used.emit(player_id, item_id, result)
	return result

## Draws one loot from the deck into the player's hand (reshuffling the
## discard when needed). Returns the card, or null if no loot is left.
func draw_loot(player_id: int) -> CardResource:
	var card := _pop_loot()
	if card == null:
		return null
	EventBus.card_drawn.emit(player_id, card.id, "loot")
	GameState.players[player_id]["hand"].append(card.id)
	# ESEMPIO 1 step 1/6: "P1 draws {card_name}"
	print("%s draws %s" % [_pname(player_id), card.display_name])
	return card

## --- Priority / stack window (ESEMPI 2, 5) -------------------------------------
## The entry on top of the stack opened a window: each player WITH an
## available response is asked in turn order starting from the ACTIVE player
## (round-robin); players with NOTHING to answer with are auto-passed (the
## window is "visible to every player that has items/loot to respond with").
## A response is a loot play (ATOMIC â€” resolved at once, no nested window) or
## an eternal item use (D6/Belial push a generic modifier ON TOP and the
## round restarts from the active player); anything else counts as a pass,
## logged per ESEMPIO 2. When ALL players pass consecutively the window
## closes and resolve_top() applies the top of the stack (LIFO).
func _run_stack_window(entry: Dictionary) -> void:
	var count := GameState.players.size()
	if count == 0:
		return
	var idx := GameState.current_player_index
	var passes := 0
	while passes < count:
		if not _player_has_response(idx, entry):
			print("%s passes priority on %s (no responses available)" % [
				_pname(idx), stack.top().get("id", "?")])
			passes += 1
			idx = (idx + 1) % count
			continue
		EventBus.priority_pending.emit(idx, entry["type"], entry.duplicate())
		var args: Array = await EventBus.priority_given
		var pid: int = args[0] if args.size() > 0 and typeof(args[0]) == TYPE_INT else idx
		var response: Dictionary = args[1] if args.size() > 1 and typeof(args[1]) == TYPE_DICTIONARY else {}
		var action: String = response.get("action", "pass")
		var acted := false
		match action:
			"loot":
				var card: CardResource = card_resource(response.get("card_id", ""))
				if card != null and _play_loot_in_priority(card, pid, response.get("target_id", "")):
					acted = true
					print("  [priority] %s responds: plays %s" % [_pname(pid), card.display_name])
			"eternal":
				var res: Dictionary = use_eternal_item(pid, response.get("payload", {}))
				if res.get("ok", false):
					acted = true
					print("  [priority] %s responds: uses %s" % [_pname(pid), res.get("item_id", "?")])
			"stop":
				print("  [priority] 'stop attack' is NOT a rule (MP07 D) â€” the attack continues")
			_:
				pass  # "pass" / "resolve" / unknown
		if acted:
			passes = 0          # a response restarts the round from the ACTIVE player
			idx = GameState.current_player_index
		else:
			# ESEMPIO 2 step 2a: "P1 passes priority on r1"
			print("%s passes priority on %s" % [_pname(pid), stack.top().get("id", "?")])
			passes += 1
			idx = (idx + 1) % count
	EventBus.priority_closed.emit(entry["type"], entry.duplicate())

## Can this player legally answer THIS stack entry right now? (Eternal item
## matching the entry type and not recharging, or a hand card the loot limits
## still allow.) Players with no legal answer are auto-passed.
func _player_has_response(pid: int, entry: Dictionary) -> bool:
	var p := GameState.players[pid]
	if p.is_empty():
		return false
	if not p["eternal_used"] and p["eternal_item_id"] != "":
		var item: CardResource = card_resource(p["eternal_item_id"])
		if item != null:
			match item.eternal_action:
				"reroll_die", "adjust_die":
					if entry.get("type", "") == "pending_roll":
						return true
				"prevent_damage":
					return true  # a shield answers any window
	if not p["hand"].is_empty():
		if pid == GameState.current_player_index:
			if p["loot_played"] < 1:
				return true
			return _character_effect_is_loot(pid) and not p["character_effect_used"]
		return _character_effect_is_loot(pid) and not p["character_effect_used"]
	return false

## Plays a loot card ATOMICALLY as a priority response for `player_id`: the
## card must be in hand and the turn's loot limit (incl. character effect)
## must allow it. Response loots NEVER open a nested window â€” they resolve
## immediately, on purpose (Phase 1 simplification).
func _play_loot_in_priority(card: CardResource, player_id: int, target_id: String) -> bool:
	if not GameState.players[player_id]["hand"].has(card.id):
		return false
	var payload := {}
	if target_id != "":
		payload = {"target_id": target_id}
	if card.type == CardResource.CardType.LOOT:
		var player := GameState.players[player_id]
		if player_id == GameState.current_player_index:
			if player["loot_played"] < 1:
				player["loot_played"] += 1
			elif _character_effect_is_loot(player_id) and not player["character_effect_used"]:
				player["character_effect_used"] = true
			else:
				return false
		else:
			if not _character_effect_is_loot(player_id) or player["character_effect_used"]:
				return false
			player["character_effect_used"] = true
	_resolve_loot_immediately(card, player_id, payload)
	GameState.players[player_id]["hand"].erase(card.id)
	return true

## Applies a loot/item card WITHOUT the stack (atomic priority responses).
func _resolve_loot_immediately(card: CardResource, player_id: int, payload: Dictionary) -> void:
	CardEffectRegistry.play_card(card, player_id, payload)
	EventBus.card_played.emit(player_id, card.id)

## --- Stack resolution (ESEMPI 2, 2b, 3, 5) --------------------------------------
## One switch resolving every entry type; wired as EventStack.resolver.
func _resolve_stack_entry(entry: Dictionary) -> Dictionary:
	match entry.get("type", ""):
		"pending_roll":
			return _resolve_pending_roll(entry)
		"modifier":
			return _resolve_modifier(entry)
		"pending_effect":
			return _resolve_pending_effect(entry)
		_:
			Log.warn("MatchController: unknown stack entry type '%s'" % str(entry.get("type", "")))
			return {}

## ESEMPIO 2 step 3 / 2b step 4: compare the (possibly modified) value with
## the monster's die value and log the canonical line.
func _resolve_pending_roll(entry: Dictionary) -> Dictionary:
	var die_value: int = entry.get("die_value", 0)
	var target: String = entry.get("target", "")
	var monster := _find_monster(target)
	if not monster.is_empty():
		die_value = monster["card"].die_value
	var value: int = entry.get("value", 0)
	var hit := value >= die_value
	print("%s attack resolves: %d vs die %d -> %s" % [
		_pname(entry.get("owner", 0)), value, die_value, "HIT" if hit else "MISS"])
	EventBus.monster_attacked.emit(entry.get("owner", 0), target, value, entry.get("bonus", 0), die_value)
	if target != "":
		CardEffectRegistry.notify_monster_attacked(target, entry.get("owner", 0), value)
	return {"hit": hit, "roll": value, "die_value": die_value}

## Modifier resolution: reroll executes NOW (deferred from activation),
## updating the targeted pending_roll; adjust was already applied at
## activation, so resolving it is a no-op confirmation.
func _resolve_modifier(entry: Dictionary) -> Dictionary:
	var target: Dictionary = stack.find(entry.get("target_id", ""))
	match entry.get("effect", ""):
		"reroll":
			var old_value: int = target.get("value", 0)
			var bonus: int = target.get("bonus", 0)
			var new_value: int = Rng.roll_die() + bonus
			target["value"] = new_value
			# ESEMPIO 2 step 3: "D6 reroll: r1 value changes from 2 to 5"
			print("%s reroll: %s value changes from %d to %d" % [
				entry.get("item_name", "Die"), entry.get("target_id", "?"), old_value, new_value])
			return {"rerolled_from": old_value, "roll": new_value}
		_:
			return {"applied": true}

## ESEMPIO 5 step 3: the loot effect applies and logs its resolution.
func _resolve_pending_effect(entry: Dictionary) -> Dictionary:
	var card: CardResource = card_resource(entry.get("card_id", ""))
	if card == null:
		Log.warn("MatchController: pending_effect references missing card '%s'" % str(entry.get("card_id", "")))
		return {"applied": false}
	var payload: Dictionary = entry.get("payload", {})
	# capture the target's DISPLAY NAME before the effect may kill & remove it
	var target_id: String = payload.get("target_id", "")
	var target := _find_monster(target_id)
	var target_name: String = target["card"].display_name if not target.is_empty() else target_id
	CardEffectRegistry.play_card(card, entry.get("owner", 0), payload)
	if card.simple_effect == "damage_monster":
		print("%s resolves: %s takes %d damage" % [card.display_name, target_name, card.simple_effect_value])
	else:
		print("%s resolves" % card.display_name)
	EventBus.card_played.emit(entry.get("owner", 0), card.id)
	return {"applied": true, "card_id": card.id}

## ESEMPIO 3 step 4 â€” called on EVERY attack exit path (kill OR player death):
## combat flag cleared, stack VERIFIED empty, ATTACK available again.
func reset_combat_state() -> void:
	_combat_in_progress = false
	if not stack.is_empty():
		Log.warn("MatchController: reset_combat_state found %d leftover entries â€” clearing" % stack.size())
		stack.clear("combat_reset")
	EventBus.combat_state_reset.emit()
	print("Combat resolved, stack empty, attack available again")

## --- Death (MP06 f / ESEMPIO 3 acceptance) --------------------------------------
## HP <= 0 is NOT match over: the player loses 1 loot from hand (if any) and
## 1 cent (if any), then chooses one owned item to lose (the eternal item is
## protected). The ending phase then resolves like any other turn end â€”
## INCLUDING advance_active_player(). Asynchronous: the driver must answer
## player_death_resolved(player_id, item_id).
func die(player_id: int) -> void:
	var player := GameState.players[player_id]
	var penalty := {"cents_lost": false, "loot_lost": "", "item_options": []}
	if player["cents"] > 0:
		player["cents"] -= 1
		penalty["cents_lost"] = true
		EventBus.cents_changed.emit(player_id, -1, player["cents"])
	if not player["hand"].is_empty():
		penalty["loot_lost"] = player["hand"][0]
		player["hand"].pop_front()
		var lost_card: CardResource = card_resource(penalty["loot_lost"])
		if lost_card != null:
			_loot_discard.append(lost_card)   # keep the loot circulating (resources)
		EventBus.card_discarded.emit(player_id, penalty["loot_lost"])
	for card_id in player["items_passive"]:
		var item: CardResource = card_resource(card_id)
		if item == null or not item.is_eternal:
			penalty["item_options"].append(card_id)
	EventBus.player_died.emit(player_id, penalty)
	var args: Array = await EventBus.player_death_resolved
	var item_id: String = args[1] if args.size() > 1 else ""
	if item_id != "" and player["items_passive"].has(item_id):
		player["items_passive"].erase(item_id)
		CardEffectRegistry.unregister_card_by_card_id(player_id, item_id)
		print("  %s loses item %s to the death penalty" % [_pname(player_id), item_id])
	_finish_turn(player_id)

## --- Table / deck internals ---------------------------------------------------

func card_resource(card_id: String) -> CardResource:
	return load(CARD_DIR + card_id + ".tres")

func is_bought(card_id: String, player_id: int) -> bool:
	return _bought.get(player_id, {}).has(card_id)

## UI-facing shim (ESEMPIO 2 general rule): the D6/Belial buttons gate on
## "does a pending_roll sit on the stack?", never on whose turn it is.
func get_pending_roll() -> Dictionary:
	var entry: Dictionary = stack.top_of_type("pending_roll") if stack != null else {}
	if entry.is_empty():
		return {}
	return {
		"id": entry["id"], "player_id": entry.get("owner", 0),
		"roll": entry.get("value", 0), "bonus": entry.get("bonus", 0),
		"die_value": entry.get("die_value", 0), "target": entry.get("target", "")}

func is_combat_in_progress() -> bool:
	return _combat_in_progress

func get_shop_card(slot_index: int) -> CardResource:
	if slot_index < 0 or slot_index >= shop_slots.size():
		return null
	return shop_slots[slot_index]["card"]

func _find_monster(instance_id: String) -> Dictionary:
	for monster in table_monsters:
		if monster["instance_id"] == instance_id:
			return monster
	return {}

func _character_effect_is_loot(player_id: int) -> bool:
	var char_card := card_resource(GameState.players[player_id]["character_id"])
	return char_card != null and char_card.character_effect == "play_loot"

## Kill rewards (MP07 J): a boss yields its OWN soul card (id + soul_value);
## other soul_reward monsters yield their color; vanilla monsters yield 1Â¢.
## Rewards come from CARD DATA â€” never hardcoded (ESEMPIO 3 step 2).
func _kill_rewards(card: CardResource) -> Dictionary:
	if card.is_boss:
		return {"soul": card.id, "soul_value": card.soul_value}
	if card.soul_reward != "":
		return {"soul": card.soul_reward}
	return {"cents": 1}

func _build_decks() -> void:
	var dir := DirAccess.open(CARD_DIR)
	if dir == null:

## Returns the number of .tres files written, or -1 on a fatal error.
static func import_from_csv(csv_path: String = DEFAULT_CSV_PATH, output_dir: String = DEFAULT_OUTPUT_DIR) -> int:
	var text := FileAccess.get_file_as_string(csv_path)
	if text.is_empty():
		push_error("CardImporter: cannot read CSV at %s" % csv_path)
		return -1
	var rows := _parse_csv(text)
	if rows.size() < 2:
		push_error("CardImporter: no data rows found in %s" % csv_path)
		return -1
	var headers: Array = rows[0]
	var card_script = load(CARD_RESOURCE_PATH)
	if card_script == null:
		push_error("CardImporter: cannot load %s" % CARD_RESOURCE_PATH)
		return -1
	var type_map := {
		"CHARACTER": card_script.CardType.CHARACTER,
		"MONSTER": card_script.CardType.MONSTER,
		"ITEM_ACTIVE": card_script.CardType.ITEM_ACTIVE,
		"ITEM_PASSIVE": card_script.CardType.ITEM_PASSIVE,
		"LOOT": card_script.CardType.LOOT,
		"TREASURE": card_script.CardType.TREASURE,
		"SOUL": card_script.CardType.SOUL,
		"CURSE": card_script.CardType.CURSE,
	}
	var dir := DirAccess.open("res://")
	if dir == null:
		push_error("CardImporter: cannot open res://")
		return -1
	dir.make_dir_recursive(output_dir.trim_prefix("res://"))
	var created := 0
	var errors: Array = []   # one entry per problem; all problems are reported
	var seen_ids := {}
	for i in range(1, rows.size()):
		var fields: Array = rows[i]
		if fields.size() != headers.size():
			errors.append("row %d: %d fields, expected %d" % [i + 1, fields.size(), headers.size()])
			continue
		var row := {}
		for j in range(headers.size()):
			row[String(headers[j]).strip_edges()] = String(fields[j]).strip_edges()
		var problems: Array = []
		var card_id: String = row.get("id", "")
		if card_id == "":
			problems.append("missing id")
		elif seen_ids.has(card_id):
			problems.append("duplicate id '%s'" % card_id)
		seen_ids[card_id] = true
		var effect_path: String = row.get("effect_script_path", "")
		if effect_path != "" and not FileAccess.file_exists(effect_path):
			problems.append("effect script not found: %s" % effect_path)
		var card = card_script.new()
		_apply_fields(card, row, type_map, problems)
		if not problems.is_empty():
			for problem in problems:
				errors.append("card '%s' (row %d): %s" % [card_id if card_id != "" else "?", i + 1, problem])
			continue
		var err := ResourceSaver.save(card, output_dir + card_id + ".tres")
		if err != OK:
			errors.append("card '%s': failed to save .tres (error %d)" % [card_id, err])
		else:
			created += 1

		print("CardImporter: %d cards written to %s, 0 errors" % [created, output_dir])
	else:
		print("CardImporter: %d cards written to %s, %d error(s):" % [created, output_dir, errors.size()])

			print("  - " + problem)

static func _apply_fields(card: Resource, row: Dictionary, type_map: Dictionary, problems: Array) -> bool:
	var type_name: String = row.get("type", "")
	if not type_map.has(type_name):
		problems.append("unknown card type '%s'" % type_name)
		return false
	card.id = row.get("id", "")
	card.display_name = row.get("display_name", "")
	card.type = type_map[type_name]
	for num_field in ["cost_cents", "die_value", "hp", "roll_bonus", "simple_effect_value", "needs_target", "playable", "is_eternal", "recharges_on_turn_end", "damage", "damage_bonus", "soul_value", "is_boss"]:
		var raw: String = row.get(num_field, "")
		if raw != "" and not raw.is_valid_int():
			problems.append("field '%s' is not an integer ('%s')" % [num_field, raw])
			return false
	card.cost_cents = int(row.get("cost_cents", ""))
	card.die_value = int(row.get("die_value", ""))
	card.hp = int(row.get("hp", ""))
	card.roll_bonus = int(row.get("roll_bonus", ""))
	card.needs_target = int(row.get("needs_target", "")) != 0
	card.playable = int(row.get("playable", "")) != 0
	card.is_eternal = int(row.get("is_eternal", "")) != 0
	card.recharges_on_turn_end = int(row.get("recharges_on_turn_end", "")) != 0
	card.damage = int(row.get("damage", ""))
	card.damage_bonus = int(row.get("damage_bonus", ""))
	card.soul_value = int(row.get("soul_value", ""))
	card.is_boss = int(row.get("is_boss", "")) != 0
	card.soul_reward = row.get("soul_reward", "")
	card.text = row.get("text", "")
	card.effect_script_path = row.get("effect_script_path", "")
	var triggers: String = row.get("trigger_events", "")
	card.trigger_events.clear()
	if triggers != "":
		for t in triggers.split(";"):
			var trimmed := String(t).strip_edges()
			if trimmed != "":
				card.trigger_events.append(trimmed)
	card.simple_effect = row.get("simple_effect", "")
	card.simple_effect_value = int(row.get("simple_effect_value", ""))
	card.eternal_action = row.get("eternal_action", "")
	card.character_effect = row.get("character_effect", "")
	card.eternal_item_id = row.get("eternal_item_id", "")
	card.description = row.get("description", "")
	card.activation_type = row.get("activation_type", "")
	card.activation_cost = row.get("activation_cost", "")
	card.effect_trigger = row.get("effect_trigger", "")
	card.effect_condition = row.get("effect_condition", "")
	card.effect_description = row.get("effect_description", "")
	# MP07 I: monsters must declare effect_trigger/effect_description EXPLICITLY
	# ("none" is allowed; a missing field is not â€” the field must exist).
	if type_name == "MONSTER":
		if card.effect_trigger == "" or card.effect_description == "":
			problems.append("monster needs effect_trigger and effect_description (explicit; use 'none' when it has no special effect)")
			return false
		if not ["none", "on_roll", "on_death", "on_miss"].has(card.effect_trigger):
			problems.append("unknown monster effect_trigger '%s' (none|on_roll|on_death|on_miss)" % card.effect_trigger)
			return false
	# MP07 H: every item must carry a readable description, categorized in one
	# of the three real activation types, with an explicit cost when paid.
	if type_name == "TREASURE" or type_name == "ITEM_ACTIVE" or type_name == "ITEM_PASSIVE":
		if card.description == "":
			problems.append("item needs a description (MP07 H)")
			return false
		if not ["passive", "activated_free", "activated_paid"].has(card.activation_type):
			problems.append("activation_type must be passive|activated_free|activated_paid ('%s')" % card.activation_type)
			return false
		if card.activation_type == "activated_paid" and card.activation_cost == "":
			problems.append("activated_paid items need an explicit activation_cost (MP07 H)")
			return false
	# MP07 J: a boss always grants souls (soul_value >= 1)
	if card.is_boss and card.soul_value < 1:
		problems.append("is_boss requires soul_value >= 1")
		return false
	return true

## Minimal CSV parser: handles double-quoted fields and CRLF/LF line endings.
static func _parse_csv(text: String) -> Array:
	var rows: Array = []
	var row: Array = []
	var field := ""
	var in_quotes := false
	var i := 0
	while i < text.length():
		var c: String = text[i]
		if in_quotes:
			if c == '"':
				if i + 1 < text.length() and text[i + 1] == '"':
					field += '"'
					i += 1
				else:
					in_quotes = false
			else:
				field += c
		elif c == '"':
			in_quotes = true
		elif c == ",":
			row.append(field)
			field = ""
		elif c == "\n" or c == "\r":
			if field != "" or not row.is_empty():
				row.append(field)
				rows.append(row)
				row = []
				field = ""
		else:
			field += c
		i += 1
	if field != "" or not row.is_empty():
		row.append(field)
		rows.append(row)
	return rows
